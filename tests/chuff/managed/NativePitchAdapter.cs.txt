using UnityEngine;
using DV.DopplerEffects;

// Adapter fields mirror LayeredAudio.Layer. Only SetPitch below is extracted native code.
public class NativeLayerPitchAdapter
{
    public bool usePitchCurve = true, useDoppler;
    public float startPitch = 1f;
    public AnimationCurve pitchCurve;
    public AudioSource source = new AudioSource();
    public Doppler doppler = new Doppler();
    /*NATIVE_LAYER_SETPITCH*/
}

public class StockLoopCase
{
    public readonly string Label;
    public readonly string ClipName;
    public readonly float SourcePitch;
    public readonly float[][] Points;
    public StockLoopCase(string label, string clipName, float pitch, float[][] points) { Label=label; ClipName=clipName; SourcePitch=pitch; Points=points; }
    public AnimationCurve NativeCurve() { var a=Points[0]; var b=Points[Points.Length-1]; return AnimationCurve.Linear(a[0],a[1],b[0],b[1]); }
    public float EvaluateLinear(float value)
    {
        // Stock pitch curves have two collinear keys. This adapter verifies their recorded line,
        // and does not claim to run Unity's Hermite/weighted curve evaluator.
        var first=Points[0]; var last=Points[Points.Length-1];
        return first[1]+(value-first[0])*(last[1]-first[1])/(last[0]-first[0]);
    }
}
